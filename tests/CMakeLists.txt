find_package(Threads REQUIRED)
add_executable(pdf_document_tests document_test.cpp)
target_link_libraries(pdf_document_tests PRIVATE HuxerUI::PDF Threads::Threads)
target_compile_definitions(pdf_document_tests PRIVATE PDF_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
huxerui_pdf_copy_runtime(pdf_document_tests)
add_test(NAME pdf.document COMMAND pdf_document_tests)
set_tests_properties(pdf.document PROPERTIES TIMEOUT 60)

if(HUXERUI_PDF_BUILD_VIEW)
    add_executable(pdf_view_tests view_test.cpp)
    target_link_libraries(pdf_view_tests PRIVATE HuxerUI::PDF)
    target_compile_definitions(pdf_view_tests PRIVATE PDF_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
    if(TARGET HuxerUI::testing)
        target_compile_definitions(pdf_view_tests PRIVATE PDF_HAS_UI_TESTING=1)
        target_link_libraries(pdf_view_tests PRIVATE HuxerUI::testing)
    endif()
    huxerui_pdf_copy_runtime(pdf_view_tests)
    add_test(NAME pdf.view COMMAND pdf_view_tests)
    set_tests_properties(pdf.view PROPERTIES TIMEOUT 60)
endif()
foreach(_target IN ITEMS pdf_document_tests pdf_view_tests)
    if(TARGET ${_target})
        target_compile_options(${_target} PRIVATE "$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
    endif()
endforeach()
